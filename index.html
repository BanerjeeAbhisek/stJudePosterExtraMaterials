methodInfo = [
  {name:"PCA",
   what:"Takes one data matrix and finds the few directions along which the samples vary the most, then rewrites each sample with coordinates along those directions instead of the original variables.",
   why:"The workhorse for dimension reduction and visualization — and the building block the others extend."},
  {name:"PLSKERN",
   what:"Given predictors X and a response Y on the same samples, it builds components of X that vary together with Y (maximizing their covariance) — supervised, unlike PCA. The 'kernel' part is just a faster route to the same fit via cross-product matrices.",
   why:"Handles many correlated variables (p ≫ n) where ordinary regression fails; the kernel form is a fast route to the same PLS fit."},
  {name:"CCA",
   what:"Takes two data blocks X and Y measured on the same samples and finds the weighted combination of each whose resulting scores are as correlated as possible — i.e. the strongest shared pattern between the two blocks.",
   why:"Quantifies how two sets of measurements on the same samples relate — the classic two-block tool."},
  {name:"PMD",
   what:"Approximates a matrix as a sum of rank-1 pieces (score × loading), but with penalties that push most loading weights to exactly zero — a low-rank factorization that is also sparse.",
   why:"The engine behind sparse PCA/CCA — gives interpretable factors by zeroing out most loadings."},
  {name:"SPC",
   what:"PCA with a sparsity penalty: each principal component is still a direction of high variance, but it's allowed to use only a small subset of variables (the rest get weight zero).",
   why:"Keeps PCA's structure but yields loadings you can actually read — a named short list per component."},
  {name:"SCCA",
   what:"CCA with sparsity: it finds the correlated directions between two blocks X and Z, but forces each direction to rely on only a few variables per block instead of all of them.",
   why:"Finds which specific variables in each block drive the association — feasible even when p ≫ n."},
  {name:"JIVE",
   what:"Takes several data types measured on the same samples and decomposes each into three parts: a joint structure shared across all types, individual structure unique to that type, and residual noise.",
   why:"Separates signal common to all sources from what's unique to each — what plain PCA/CCA can't do."},
   {name:"PLS-DA",
   what:"PLS turned into a classifier: the response is the class label (one-hot encoded), so it finds the directions of X that best separate the known groups. Uses all variables — the dense counterpart of sPLS-DA.",
   why:"Supervised group separation and prediction when variables are many and correlated (p ≫ n)."},
  {name:"sPLS-DA",
   what:"PLS used for classification: the response is the class label (one-hot encoded), and a sparsity penalty makes each component depend on only a few variables — so it separates the classes and names the variables doing it.",
   why:"Predicts the class and names the few variables responsible — interpretable biomarker selection."}
]

    "PMD":      {"PCA":"adds sparsity penalty", "PLSKERN":"matrix factori zation, not regression", "CCA":"penalized, sparse factors", "SPC":"general engine (SPC is a use of it)", "SCCA":"general engine (SCCA uses it)", "JIVE":"single matrix", "PLS-DA":"unsuper vised, no classes", "sPLS-DA":"unsupervised, no classes"},

    "SCCA":     {"PCA":"two blocks + sparse", "PLSKERN":"correlation + sparse", "CCA":"adds variable selection", "PMD":"specific case (sparse CCA)", "SPC":"two blocks, not one", "JIVE":"shared only, sparse", "PLS-DA":"unsupervised two-block", "sPLS-DA":"unsupervised X–Z"},
    "JIVE":     {"PCA":"≥2 types, splits joint/indiv", "PLSKERN":"≥2 types, no response", "CCA":"adds individual structure", "PMD":"multi-type decom position", "SPC":"multi-type, not sparse", "SCCA":"joint + individual (not just shared)", "PLS-DA":"unsuper vised integration", "sPLS-DA":"unsupervised integration"},
    "PLS-DA":   {"PCA":"supervised — uses class labels", "PLSKERN":"classifies (class Y), not regression", "CCA":"supervised classi fication", "PMD":"supervised, uses classes", "SPC":"supervised, uses classes", "SCCA":"classify, not correlate", "JIVE":"classification, single X", "sPLS-DA":"dense — uses all variables"},
